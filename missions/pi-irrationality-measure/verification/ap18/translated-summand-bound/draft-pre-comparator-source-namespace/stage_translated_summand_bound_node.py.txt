         'AP18 final contexts await actual Published multiplicity and genuine closed Accepted sum-weight certificate')

    for cert_path,base,consumer in certificates:

def check_only():
    outputs=final_files()
    for f,raw in outputs.items():need((STAGE/f).read_bytes()==raw,'Immutable AP18 final file differs: '+f)
    return {'checked_files':len(outputs),'actual_prerequisites_replayed':True,'Lean_started':False,'API_used':False}

def write_exact(outputs):
    for f,raw in outputs.items():
        path=STAGE/f
        if path.exists():need(path.read_bytes()==raw,'Refusing overwrite: '+f)
        else:path.parent.mkdir(parents=True,exist_ok=True);path.write_bytes(raw)
    for f,raw in outputs.items():need((STAGE/f).read_bytes()==raw,'Written pure output differs: '+f)

def main():
    p=argparse.ArgumentParser(description=__doc__);g=p.add_mutually_exclusive_group(required=True)
    for flag in ['prepare-candidate','check-candidate','bind-published','finalize','check-only']:g.add_argument('--'+flag,action='store_true')
    a=p.parse_args()
    if a.bind_published:result=bind_published()
    elif a.finalize:
        outputs=final_files();write_exact(outputs);result={'prepared':True,'files':len(outputs),'compile_commands':read(STAGE/'preparation.json')['compile_commands'],'API_used':False,'Lean_started':False}
    elif a.check_only:result=check_only()
    else:
        if a.prepare_candidate:write_exact(candidate_files())
        result=check_candidate()
    print(json.dumps(result,ensure_ascii=False,indent=2))
if __name__=='__main__':main()
